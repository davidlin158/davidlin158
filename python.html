from pathlib import Path
import csv, json
from collections import defaultdict

def summarize(path: Path) -> dict:
    """读 CSV，返回每列的取值计数。坏行跳过并记录。"""
    counts, bad = defaultdict(lambda: defaultdict(int)), []
    with path.open(encoding="utf-8", newline="") as f:
        for lineno, row in enumerate(csv.DictReader(f), start=2):
            try:
                for col, val in row.items():
                    counts[col][val.strip() or "<缺失>"] += 1
            except AttributeError:
                bad.append(lineno)
    return {"counts": {k: dict(v) for k, v in counts.items()}, "bad_lines": bad}

if __name__ == "__main__":
    print(json.dumps(summarize(Path("data/raw.csv")), ensure_ascii=False, indent=2))

import time, functools
from dataclasses import dataclass, field
from typing import Callable, Iterator

def timed(fn: Callable) -> Callable:
    @functools.wraps(fn)                 # 保住原函数的名字和 docstring
    def wrapper(*args, **kwargs):
        t0 = time.perf_counter()
        out = fn(*args, **kwargs)
        print(f"{fn.__name__} 用了 {time.perf_counter() - t0:.3f}s")
        return out
    return wrapper

@dataclass(frozen=True)                  # frozen=True 防止配置被偷偷改掉
class CleanConfig:
    drop_threshold: float = 0.5
    date_cols: list[str] = field(default_factory=list)

@timed
def read_lines(path: str) -> Iterator[str]:   # 生成器：不把整个文件读进内存
    with open(path, encoding="utf-8") as f:
        for line in f:
            yield line.rstrip("\n")

import numpy as np

from statsmodels.stats.power import NormalIndPower
from statsmodels.stats.proportion import proportion_effectsize, proportions_ztest

# 1) 开跑之前先算样本量，不是跑完再算
effect = proportion_effectsize(0.032, 0.030)          # 3.0% → 3.2%
n = NormalIndPower().solve_power(effect, power=0.8, alpha=0.05, ratio=1)
print(f"每组至少需要 {np.ceil(n):,.0f} 个样本")

# 2) 结果检验
conv = np.array([1_602, 1_501]); n_obs = np.array([50_000, 50_000])
z, p = proportions_ztest(conv, n_obs)

# 3) 只报 p 值是不够的，必须给效应量的区间
rng = np.random.default_rng(42)
a = rng.binomial(1, conv[0]/n_obs[0], n_obs[0]); b = rng.binomial(1, conv[1]/n_obs[1], n_obs[1])
diffs = [rng.choice(a, a.size).mean() - rng.choice(b, b.size).mean() for _ in range(2_000)]
lo, hi = np.percentile(diffs, [2.5, 97.5])
print(f"p={p:.4f}  提升 {np.mean(diffs)*100:+.3f}pp  95% CI [{lo*100:+.3f}, {hi*100:+.3f}]pp")

from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.impute import SimpleImputer
from sklearn.model_selection import cross_val_score, StratifiedKFold
from lightgbm import LGBMClassifier

num = ["amount", "tenure_days"]; cat = ["city", "channel"]

pre = ColumnTransformer([
    ("num", Pipeline([("imp", SimpleImputer(strategy="median")),
                      ("sc",  StandardScaler())]), num),
    ("cat", OneHotEncoder(handle_unknown="ignore", min_frequency=20), cat),
])

model = Pipeline([("pre", pre), ("clf", LGBMClassifier(n_estimators=400, learning_rate=0.05))])

cv = StratifiedKFold(5, shuffle=True, random_state=42)
scores = cross_val_score(model, X, y, cv=cv, scoring="average_precision")
print(f"PR-AUC {scores.mean():.4f} ± {scores.std():.4f}")

import joblib, logging
from fastapi import FastAPI
from pydantic import BaseModel, Field

log   = logging.getLogger(__name__)
model = joblib.load("artifacts/model.joblib")     # Pipeline，预处理已在里面
app   = FastAPI(title="churn-scorer", version="1.0.0")

class Order(BaseModel):                            # 入参校验，脏数据在门口就被挡住
    amount: float = Field(ge=0)
    tenure_days: int = Field(ge=0)
    city: str
    channel: str

@app.post("/predict")
def predict(order: Order) -> dict:
    X = pd.DataFrame([order.model_dump()])
    p = float(model.predict_proba(X)[0, 1])
    log.info("scored", extra={"city": order.city, "p": round(p, 4)})
    return {"churn_probability": round(p, 4), "model_version": app.version}

@app.get("/health")                                # 部署系统靠它判断你还活着
def health() -> dict:
    return {"status": "ok"}

新手十坑

Ten traps

这十个坑会吃掉你前两个月里最多的时间。提前看一遍，踩到时能立刻认出来。

01 · df[df.a > 1]["b"] = 0 悄悄没生效

链式赋值改的是副本。用 df.loc[df.a > 1, "b"] = 0，或者干脆用 .assign() 生成新表。

02 · 先标准化，再切分训练测试集

测试集的统计量泄漏进了训练。把所有预处理放进 Pipeline，让每一折自己 fit。

03 · 用未来数据造特征

滚动、排名、聚合时问一句「这个值在预测时刻真的拿得到吗」。时间数据一律 TimeSeriesSplit。

04 · 不平衡数据只看 accuracy 或 AUC

1% 正样本时全猜负也有 99% 准确率。看 PR-AUC、混淆矩阵，并明确业务更怕漏还是更怕错。

05 · 没有基线就说模型「效果不错」

先跑 DummyClassifier 或「用昨天的值预测今天」。比不过基线的复杂模型没有价值。

06 · merge 之后行数变了却没发现

养成 validate="many_to_one" + indicator=True 的习惯，join 前后各打一次 len(df)。

07 · 可变默认参数 def f(x, acc=[])

默认值在定义时创建一次，会跨调用累积。用 acc=None 再在函数体里初始化。

08 · 在大 DataFrame 上滥用 apply

apply 是伪装成向量化的 for 循环。优先内置聚合、map、np.where、merge。

09 · 没固定随机种子，结果复现不出来

切分、模型、采样各自都有 seed。写进 config，并在 README 里写明版本。

10 · 所有代码都堆在一个 notebook 里

跑通一段就搬进 src/ 写成函数，notebook 只留调用和图。别人复现不了的分析等于没做。
